    <script>
        function calculatePayments() {
            // Obtener valores del formulario
            const price = parseFloat(document.getElementById('price').value);
            const plan = parseInt(document.getElementById('plan').value);

            if (isNaN(price) || price <= 0) {
                document.getElementById('result').innerHTML = "<p style='color: red;'>Por favor, ingresa un precio válido.</p>";
                return;
            }

            let discountRate, downPaymentRate, months, downPayment, remainingBalance, monthlyPayment, totalAfterDiscount;

            // Calcular según el plan seleccionado
            if (plan === 12) {
                discountRate = 0.20;
                downPaymentRate = 0.50;
                months = 12;
            } else {
                discountRate = 0.15;
                downPaymentRate = 0.20;
                months = 18;
            }

            // Aplicar descuento
            totalAfterDiscount = price * (1 - discountRate);

            // Calcular enganche
            downPayment = totalAfterDiscount * downPaymentRate;

            // Calcular saldo restante
            remainingBalance = totalAfterDiscount - downPayment;

            // Calcular mensualidades
            monthlyPayment = remainingBalance / months;

            // Generar tabla de resultados
            const resultDiv = document.getElementById('result');
            resultDiv.innerHTML = `
                <h3>Esquema de Pago</h3>
                <table>
                    <tr>
                        <th>Concepto</th>
                        <th>Monto (MXN)</th>
                    </tr>
                    <tr>
                        <td>Precio original</td>
                        <td>$${price.toLocaleString('es-MX')}</td>
                    </tr>
                    <tr>
                        <td>Descuento (${discountRate * 100}%)</td>
                        <td>$${(price * discountRate).toLocaleString('es-MX')}</td>
                    </tr>
                    <tr>
                        <td>Precio con descuento</td>
                        <td>$${totalAfterDiscount.toLocaleString('es-MX')}</td>
                    </tr>
                    <tr>
                        <td>Enganche (${downPaymentRate * 100}%)</td>
                        <td>$${downPayment.toLocaleString('es-MX')} (a pagar en los primeros 90 días)</td>
                    </tr>
                    <tr>
                        <td>Saldo</td>
                        <td>$${remainingBalance.toLocaleString('es-MX')}</td>
                    </tr>
                    <tr>
                        <td>Mensualidades (${months} meses)</td>
                        <td>$${monthlyPayment.toLocaleString('es-MX')} por mes</td>
                    </tr>
                </table>
            `;
        }
    </script>
